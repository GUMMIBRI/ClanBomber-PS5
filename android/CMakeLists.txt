cmake_minimum_required(VERSION 3.22)
project(ClanBomberAndroid LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(${CLAN_DEPS_ROOT}/SDL2 ${CMAKE_BINARY_DIR}/sdl)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
set(SDL2IMAGE_PNG_SAVE OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_JPG_SAVE OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_VENDORED OFF CACHE BOOL "" FORCE)
add_subdirectory(${CLAN_DEPS_ROOT}/SDL2_image ${CMAKE_BINARY_DIR}/image)
set(SDL2TTF_VENDORED ON CACHE BOOL "" FORCE)
set(SDL2TTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
add_subdirectory(${CLAN_DEPS_ROOT}/SDL2_ttf ${CMAKE_BINARY_DIR}/ttf)
set(SDL2MIXER_VENDORED ON CACHE BOOL "" FORCE)
set(SDL2MIXER_DEPS_SHARED OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_FLAC OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MP3 OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_OPUS OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_VORBIS OFF CACHE STRING "" FORCE)
set(SDL2MIXER_WAVPACK OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MIDI OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MOD ON CACHE BOOL "" FORCE)
set(SDL2MIXER_MOD_XMP ON CACHE BOOL "" FORCE)
set(SDL2MIXER_MOD_XMP_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(${CLAN_DEPS_ROOT}/SDL2_mixer ${CMAKE_BINARY_DIR}/mixer)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${CLAN_SOURCE_ROOT}/upstream/src/*.cpp")
add_library(main SHARED ${GAME_SOURCES} ${CLAN_SOURCE_ROOT}/desktop/desktop_paths.cpp)
target_compile_features(main PRIVATE cxx_std_17)
target_include_directories(main PRIVATE ${CLAN_SOURCE_ROOT}/src ${CLAN_SOURCE_ROOT}/upstream/src)
target_compile_definitions(main PRIVATE CLANBOMBER_DESKTOP ENABLE_NLS=0 PACKAGE="clanbomber2" PACKAGE_VERSION="2.0" CB_DATADIR="ClanBomber2" CB_LOCALEDIR="ClanBomber2/locale")
target_link_libraries(main PRIVATE SDL2::SDL2 SDL2_image::SDL2_image SDL2_mixer::SDL2_mixer SDL2_ttf::SDL2_ttf log android)
