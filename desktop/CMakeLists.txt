cmake_minimum_required(VERSION 3.20)
project(ClanBomberDesktop LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
pkg_check_modules(SDL2_MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../upstream/src/*.cpp")
add_executable(ClanBomber ${GAME_SOURCES} desktop_paths.cpp)
target_compile_features(ClanBomber PRIVATE cxx_std_17)
target_include_directories(ClanBomber PRIVATE ../src ../upstream/src)
target_compile_definitions(ClanBomber PRIVATE CLANBOMBER_DESKTOP ENABLE_NLS=0 PACKAGE="clanbomber2" PACKAGE_VERSION="2.0" CB_DATADIR="ClanBomber2" CB_LOCALEDIR="ClanBomber2/locale")
target_link_libraries(ClanBomber PRIVATE PkgConfig::SDL2 PkgConfig::SDL2_IMAGE PkgConfig::SDL2_MIXER PkgConfig::SDL2_TTF)
